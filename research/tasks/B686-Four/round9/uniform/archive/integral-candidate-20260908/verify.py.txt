from pathlib import Path
import os, sys, subprocess, hashlib, json, time
from datetime import datetime, timezone

HERE = Path(__file__).resolve().parent
REPO = HERE.parents[4]
LEAN = REPO / '.tools/elan/toolchains/leanprover--lean4---v4.33.1/bin/lean.exe'
ENV = dict(os.environ)
ENV['LEAN_PATH'] = os.pathsep.join(map(str, [
    REPO / 'research/tasks/B686-Four/formalization/PrimeCounting', REPO,
    REPO / '.lake/build/lib/lean'] +
    [p / '.lake/build/lib/lean' for p in sorted((REPO / '.lake/packages').iterdir()) if p.is_dir()]))
ENV['TEMP'] = ENV['TMP'] = str(REPO / '.tools/tmp')
OUT = HERE / 'verification'
OUT.mkdir(exist_ok=True)
RECORDS = []

def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def run(label, command, source=None):
    start = time.monotonic()
    proc = subprocess.run(command, cwd=REPO, env=ENV, stdout=subprocess.PIPE,
        stderr=subprocess.STDOUT, text=True, encoding='utf-8', errors='replace')
    log = OUT / f'{label}.log'
    log.write_text(proc.stdout, encoding='utf-8')
    RECORDS.append(dict(label=label, command=list(map(str,command)), exit_code=proc.returncode,
        seconds=round(time.monotonic()-start,2), log=str(log.relative_to(REPO)),
        log_sha256=digest(log), source=str(source.relative_to(REPO)) if source else None,
        source_sha256=digest(source) if source else None))
    print(f'{label}: exit {proc.returncode}', flush=True)
    if proc.returncode:
        print(proc.stdout,flush=True)
    return proc.returncode == 0

started = datetime.now(timezone.utc).isoformat()
ok = run('source-policy',[sys.executable,str(REPO/'scripts/check-lean-policy.py'),'--roots',str(HERE)])
ok = run('lean-version',[str(LEAN),'--version']) and ok
for name in ['Algebraic','Constants','Choose','WeightedPrime','UniformDistance','Consumer']:
    if not ok:
        break
    source = HERE / f'{name}.lean'
    ok = run(name,[str(LEAN),'-DautoImplicit=false','-DrelaxedAutoImplicit=false',
        '-o',str(HERE/f'{name}.olean'),str(source)],source)

frozen = [
    'formalization/UniformDistance/Boundary.lean',
    'formalization/UniformDistance/BigPrimeBinomial.lean',
    'formalization/UniformDistance/SmallPrime.lean',
    'formalization/UniformDistance/GcdBound.lean',
    'formalization/PrimeCounting/LinearPrimeCounting.lean',
    'round8/continuation/SizeBounds.lean',
    'independent/TargetBridge.lean',
    'round8/lean/BigPrimeSupport.lean']
base = REPO / 'research/tasks/B686-Four'
manifest = json.loads((REPO/'lake-manifest.json').read_text(encoding='utf-8-sig'))
mathlib = next(p for p in manifest['packages'] if p['name']=='mathlib')
report = dict(success=ok, started_utc=started, finished_utc=datetime.now(timezone.utc).isoformat(),
    full_uniform_distance_bound_lean_verified=ok,
    original_B686_nonexistence_proved=False, independent_kernel_checker_used=False,
    toolchain=(REPO/'lean-toolchain').read_text().strip(), mathlib_revision=mathlib['rev'],
    method='Direct pinned Lean executable; changed sources recompiled in dependency order; exact transitive-axiom #guard_msgs; separate importing consumer',
    allowed_transitive_axioms=['propext','Classical.choice','Quot.sound'],
    accepted_roots=['B686UniformRound9.uniform_distance_bound',
        'B686UniformRound9.uniform_distance_bound_products',
        'B686UniformRound9.Consumer.original_statement',
        'B686UniformRound9.Consumer.rational_ratio_statement'] if ok else [],
    frozen_inputs=[dict(source=str((base/p).relative_to(REPO)),source_sha256=digest(base/p)) for p in frozen],
    records=RECORDS)
(HERE/'verification.json').write_text(json.dumps(report,ensure_ascii=False,indent=2),encoding='utf-8')
sys.exit(0 if ok else 1)
