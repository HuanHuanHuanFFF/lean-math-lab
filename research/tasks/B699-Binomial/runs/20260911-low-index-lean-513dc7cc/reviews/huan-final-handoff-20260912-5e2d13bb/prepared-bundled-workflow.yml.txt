        run: |
          python3 - <<'PY'
          from pathlib import Path
          import shutil, subprocess
          for name in ('memory.max', 'memory.current', 'cpu.max'):
              path = Path('/sys/fs/cgroup') / name
              print(name, path.read_text().strip() if path.is_file() else 'unavailable')
          print('disk', shutil.disk_usage('.'))
          print('meminfo', [line for line in Path('/proc/meminfo').read_text().splitlines()
                            if line.startswith(('MemTotal:', 'MemAvailable:'))])
          print(subprocess.run(['ps', '-eo', 'pid,ppid,rss,comm', '--sort=-rss'],
                               capture_output=True, text=True, check=False).stdout[:1800])
          PY

          build: 'false'

          lint: 'false'
          use-mathlib-cache: 'false'
          use-github-cache: 'false'

          python3 "$B699_RUN/verification/runner/verify_incremental.py"

          python3 "$B699_RUN/verification/runner/cache_bundle.py" export
          --root "$B699_SHARED_ROOT" --archive "$B699_RAW/ci-shared/shared.zip"

        with:
          name: b699-huan-shared-${{ github.run_id }}
          path: ${{ env.B699_RAW }}/ci-shared/shared.zip
          include-hidden-files: true
          retention-days: 3
          if-no-files-found: error
