/-! UNCOMPILED CANDIDATE. Apply the accepted product selector to
f(p)=M*primeComponent n p, cap=M*n, threshold=n. Actual U is expanded by
an accepted source identity, never supplied as a decomposition hypothesis. -/

set_option relaxedAutoImplicit false

theorem two_actual_components_above_capacity {n j H M : ℕ}
    (hM : 1 ≤ M) (hij : 11 < j) (hjn : j ≤ n / 2) (hlarge : 110 ≤ n) (hnH : n < H)
    (hcertificate : (2 * Nat.factorial 11) ^ 11 * H ^ 7 ≤ i11K * M ^ 33)
    (hno : ¬ Common n 11 j) :
    ∃ p q : ℕ, p.Prime ∧ q.Prime ∧ p ∈ ({2, 3, 5, 7} : Finset ℕ) ∧
      q ∈ ({2, 3, 5, 7} : Finset ℕ) ∧ p ≠ q ∧
      n < M * primeComponent n p ∧ n < M * primeComponent n q := by
  classical
  let S := (Finset.range 11).filter Nat.Prime
  let f : ℕ → ℕ := fun p => M * primeComponent n p
  have hcard : S.card = 4 := i11_smallPrimeCount
  have hS : S.Nonempty := Finset.card_pos.mp (by rw [hcard]; decide)
  have hsmall : S.prod (primeComponent n) = smallPrimePart n 11 := by
    simpa only [S, primeComponent] using Eq.symm (small_prime_part_eq_prod_small_primes n 11)
  have hproduct : S.prod f = M ^ 4 * smallPrimePart n 11 := by
    dsimp only [f]
    rw [Finset.prod_mul_distrib]
    simp only [Finset.prod_const, hcard]
    rw [hsmall]
  have hcap : ∀ p ∈ S, f p ≤ M * n := by
    intro p hp
    have hprime : p.Prime := (Finset.mem_filter.mp hp).2
    exact Nat.mul_le_mul_left M (primeComponent_bounds (by omega : 11 ≤ n) hprime).2
  have hU := actual_small_part_strict hij hjn hlarge hnH hcertificate hno
  have hprod : (M * n) * n ^ (S.card - 1) < S.prod f := by
    rw [hcard, hproduct]
    change (M * n) * n ^ 3 < M ^ 4 * smallPrimePart n 11
    calc
      _ = M * n ^ 4 := by rw [Nat.pow_succ n 3]; ring
      _ < M * (M ^ 3 * smallPrimePart n 11) :=
        Nat.mul_lt_mul_of_pos_left hU (by omega)
      _ = M ^ 4 * smallPrimePart n 11 := by rw [Nat.pow_succ M 3]; ring
  obtain ⟨p, hp, q, hq, hpq, hpb, hqb⟩ := exists_two_gt_of_prod_gt hS hcap hprod
  have hpPrime : p.Prime := (Finset.mem_filter.mp hp).2
  have hqPrime : q.Prime := (Finset.mem_filter.mp hq).2
  have hpMem : p ∈ ({2, 3, 5, 7} : Finset ℕ) := by
    simpa only [S, i11_small_primes] using hp
  have hqMem : q ∈ ({2, 3, 5, 7} : Finset ℕ) := by
    simpa only [S, i11_small_primes] using hq
  exact ⟨p, q, hpPrime, hqPrime, hpMem, hqMem, hpq, hpb, hqb⟩

end B699LowIndex.I11BoundedCofactors
