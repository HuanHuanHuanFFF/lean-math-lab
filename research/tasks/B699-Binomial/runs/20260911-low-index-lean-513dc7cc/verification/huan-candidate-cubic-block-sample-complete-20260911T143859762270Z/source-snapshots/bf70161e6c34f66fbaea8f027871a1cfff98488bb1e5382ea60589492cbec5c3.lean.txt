/-!
# Complete cubic exponent-block exclusion

UNCOMPILED CANDIDATE. All a in the block and every b>=0 are covered.
The input bases need only be >=2; the supplied modular inverse carries
all the coprimality information needed by the source argument.
-/

set_option autoImplicit false
set_option relaxedAutoImplicit false

namespace Math.B699.CubicExponentBlock

/-- The complete block theorem of the adopted report, Section 5.
The residue condition includes d=0 and both endpoints of [-w,w]. -/
theorem cubic_exponent_block_excludes
    {p q u L b0 w v : ℕ} (hp : 2 ≤ p) (hq : 2 ≤ q)
    (_hu : 1 ≤ u) (hb0 : 1 ≤ b0) (_hPw : w < p ^ u)
    (hQ : (q ^ b0) ^ 3 ≤ (p ^ u - w) ^ 2)
    (hinv : (p ^ u * v) % (q ^ b0) = 1)
    (hresidues : ∀ d : ℤ, -(w : ℤ) ≤ d → d ≤ (w : ℤ) →
      p ^ (u + 3 * L) < (positiveResidue (q ^ b0) v d) ^ 2)
    {a b A C : ℕ} (hua : u ≤ a) (hau : a ≤ u + L)
    (hA : 1 ≤ A) (hC : 1 ≤ C)
    (hdist : Nat.dist (p ^ a * A) (q ^ b * C) ≤ w)
    (hsmallA : A ^ 3 ≤ min (p ^ a * A) (q ^ b * C))
    (hsmallC : C ^ 3 ≤ min (p ^ a * A) (q ^ b * C)) : False := by
  have hp0 : 0 < p := by omega
  have hq1 : 1 < q := by omega
  have hbb : b0 ≤ b := second_exponent_ge hp0 hq1 hua hA hC hdist hsmallC hQ
  let X : ℕ := p ^ (a - u) * A
  let T : ℕ := q ^ (b - b0) * C
  let d : ℤ := ((p ^ a * A : ℕ) : ℤ) - ((q ^ b * C : ℕ) : ℤ)
  have hX : 0 < X := by
    dsimp [X]
    exact Nat.mul_pos (Nat.pow_pos hp0 _) (by omega)
  have hQ1 : 1 < q ^ b0 := by
    have hqle : q ≤ q ^ b0 := by
      simpa only [pow_one] using Nat.pow_le_pow_right (by omega : 0 < q) hb0
    omega
  have hPX : p ^ u * X = p ^ a * A := shifted_power_mul p a u A hua
  have hQT : q ^ b0 * T = q ^ b * C := shifted_power_mul q b b0 C hbb
  have heq : ((p ^ u : ℕ) : ℤ) * (X : ℤ) -
      ((q ^ b0 : ℕ) : ℤ) * (T : ℤ) = d := by
    have h := congrArg₂ (fun r s : ℕ => (r : ℤ) - (s : ℤ)) hPX hQT
    dsimp only [d]
    simpa only [Nat.cast_mul] using h
  have hd : -(w : ℤ) ≤ d ∧ d ≤ (w : ℤ) := signed_difference_bounds hdist
  have hrho : positiveResidue (q ^ b0) v d ≤ X :=
    inverse_positiveResidue_le hQ1 hX hinv heq
  have hXbound : X ^ 2 ≤ p ^ (u + 3 * L) :=
    extracted_square_le hp0 hua hau (by omega : 0 < A)
      (hsmallA.trans (min_le_left _ _))
  have hsmallrho : (positiveResidue (q ^ b0) v d) ^ 2 ≤ p ^ (u + 3 * L) :=
    (Nat.pow_le_pow_left hrho 2).trans hXbound
  exact (not_lt_of_ge hsmallrho) (hresidues d hd.1 hd.2)

end Math.B699.CubicExponentBlock
