procedure Main()
  SetSeed(69903);
  QQ<q> := PolynomialRing(Rationals());
  K<th> := NumberField(q^3-27*q+4);
  E := EllipticCurve([K|0,-60,0,-8700,0]);
  tau := -9*th^2+4*th+209;
  nu := 81*th^2-36*th-2171;
  Qinf := E![tau,nu,1];
  assert tau eq 290*(4-th)/(9*th+4);
  assert nu eq -11600/(9*th+4);
  R<D,W> := PolynomialRing(K,2);
  F := W^3-27*W*D^2+4*D^3+18*D*W+8*D^2-9*W-4*D;
  xp := (3*th^2-27)*W^2+(54*th-12)*D*W+(729-27*th^2-12*th)*D^2
         +(-18*th-8)*W+(18*th^2-16*th-486)*D+tau;
  yp := (Derivative(F,2)*Derivative(xp,1)-Derivative(F,1)*Derivative(xp,2))/2;
  I := ideal<R|F>;
  assert NormalForm(yp^2-xp^3+60*xp^2+8700*xp,I) eq 0;
  FR := FieldOfFractions(R);
  oldx := FR!(290*(4*D-W))/(4*D+9*W);
  oldy := FR!(-11600*(D-1))/(4*D+9*W);
  sl := (oldy+nu)/(oldx-tau);
  nx := sl^2+60-oldx-tau;
  ny := -oldy+sl*(oldx-nx);
  assert NormalForm(R!Numerator(nx-xp),I) eq 0;
  assert NormalForm(R!Numerator(ny-yp),I) eq 0;
  for coeff in Coefficients(xp) do for aa in Eltseq(coeff) do assert Denominator(aa) eq 1; end for; end for;
  for coeff in Coefficients(yp) do for aa in Eltseq(coeff) do assert Denominator(aa) eq 1; end for; end for;
  print "GLOBAL_TRANSLATION_MAP_AND_INTEGRAL_COEFFICIENTS",true;
  print "NO_SCALING_REQUIRED",true;
  print "Y_POLYNOMIAL",yp;
  A := E![-290/9,11600/27,1]; B := E![-60,-300,1];
  points := [A,B,A+B,A-B,2*A,2*B,2*A+B,A+2*B,2*A-B,A-2*B];
  for P in points do
    tt := P[1]/P[3]; vv := P[2]/P[3]; den := vv+9*tt+290;
    xx := (vv+18*tt+580)/den; uu := (tt+58)/den;
    dd := xx-1; ww := 50*uu-6*dd;
    PP := P-Qinf;
    assert Evaluate(xp,[dd,ww]) eq PP[1]/PP[3];
    assert Evaluate(yp,[dd,ww]) eq PP[2]/PP[3];
  end for;
  print "SAMPLE_TRANSLATION_CHECKS",#points;
  T, tm := TorsionSubgroup(E);
  print "TORSION_ORDER",#T;
  print "QINF_NONTORSION",(#T)*Qinf ne E!0;
  // The trace-zero relation follows from the original infinity line divisor, not this computation.
  print "INPUT_SUBGROUP_GENERATORS",A,B,Qinf;
  print "STAGE02_DIRECT_MAP_PASSED";
end procedure;
